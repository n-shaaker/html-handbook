۱۷) شرط‌ها (If...Else)

برنامه بدون شرط همیشه یک مسیر را اجرا می‌کند. با if می‌گویید: اگر این شرط درست بود، دستورهای داخل آکولاد اول را اجرا کن. else مسیر جایگزین است و فقط وقتی اجرا می‌شود که شرط غلط باشد. شرط باید عبارتی باشد که به درست یا غلط تعبیر شود؛ معمولاً همان مقایسه‌هایی که در درس قبل دیدید.

let hour = 14;

if (hour < 12) {
  console.log("صبح بخیر");
} else {
  console.log("عصر بخیر");     // این اجرا می‌شود، چون 14 < 12 غلط است
}





مقدار hour برابر ۱۴ است. شرط hour < 12 غلط است، پس بلوک اول رد می‌شود و فقط بلوک else اجرا می‌شود. اگر ساعت را به عددی کوچک‌تر از ۱۲ عوض کنید، پیام صبح چاپ می‌شود و بلوک دوم اصلاً اجرا نمی‌شود. در هر اجرا فقط یکی از این دو مسیر انتخاب می‌شود.



else if — چند حالت



let score = 16;

if (score >= 17) {
  console.log("عالی");
} else if (score >= 14) {
  console.log("خوب");
} else if (score >= 10) {
  console.log("قبول");
} else {
  console.log("مردود");
}










else if یک شرط تازه را فقط وقتی بررسی می‌کند که شرط‌های قبلی غلط بوده باشند. نمرهٔ ۱۶ از ۱۷ کوچک‌تر است، پس شاخهٔ «عالی» رد می‌شود. شرط بعدی، بزرگ‌تر یا مساوی ۱۴، درست است و «خوب» چاپ می‌شود. شرط‌های پایین‌تر دیگر دیده نمی‌شوند، حتی اگر آن‌ها هم از نظر ریاضی درست باشند.



شرط‌ها از بالا به پایین بررسی می‌شوند و اولین شرط درست اجرا می‌شود؛ بقیه نادیده گرفته می‌شوند. پس ترتیب شرط‌ها مهم است. نمره ۱۶ هم از ۱۰ بزرگ‌تر است، ولی چون شرط score >= 14 زودتر درست شده، فقط «خوب» چاپ می‌شود.



اشتباه رایج: نوشتن شرط‌ها به ترتیب اشتباه: اگر اول score >= 10 را بنویسید، نمره ۱۹ هم «قبول» می‌شود و هرگز به «عالی» نمی‌رسد. از سخت‌ترین شرط شروع کنید.



مثال کاربردی: هزینه ارسال سفارش



function shippingCost(cartTotal, city) {
  if (cartTotal >= 500000) {
    return 0;                          // خرید بالای ۵۰۰ هزار تومان: ارسال رایگان
  } else if (city === "تهران") {
    return 30000;
  } else {
    return 45000;
  }
}

console.log(shippingCost(650000, "شیراز"));   // 0
console.log(shippingCost(200000, "تهران"));   // 30000
console.log(shippingCost(200000, "تبریز"));   // 45000










تابع دو ورودی می‌گیرد: مبلغ سبد و نام شهر. اگر مبلغ به آستانه رسیده باشد، همان اول صفر برمی‌گرداند و بقیهٔ شرط‌ها بررسی نمی‌شوند. فقط وقتی خرید از آستانه کمتر باشد، شهر بررسی می‌شود. تهران یک هزینه دارد و هر شهر دیگری هزینهٔ شاخهٔ آخر را می‌گیرد. return هم نتیجه را برمی‌گرداند و هم اجرای تابع را تمام می‌کند.



وقتی داخل تابع به return می‌رسیم، تابع همان‌جا تمام می‌شود؛ برای همین می‌شد elseها را هم ننوشت. ولی برای شروع، نوشتن کامل else if خواناتر است.



مثال: کارنامه زنده



const scoreInput = document.getElementById("score-input");
const checkBtn = document.getElementById("check-btn");
const message = document.getElementById("score-msg");

checkBtn.addEventListener("click", function () {
  const score = Number(scoreInput.value);

  if (scoreInput.value === "") {
    message.textContent = "لطفاً یک نمره وارد کنید";
  } else if (score < 0 || score > 20) {
    message.textContent = "نمره باید بین ۰ تا ۲۰ باشد";
  } else if (score >= 17) {
    message.textContent = "عالی 🎉";
  } else if (score >= 10) {
    message.textContent = "قبول";
  } else {
    message.textContent = "مردود";
  }
});

اول متن خام کادر بررسی می‌شود، چون کادر خالی با Number به صفر تبدیل می‌شود و اگر این بررسی نباشد، خالی بودن با نمرهٔ صفر قاطی می‌شود. بعد بازه چک می‌شود تا عدد بیرون از ۰ تا ۲۰ پیام جدا بگیرد. فقط اگر هر دو مشکل نبود، نمره با آستانه‌های عالی و قبول مقایسه می‌شود.



ویژگی‌های min و max در HTML فقط پیشنهاد هستند و کاربر باز هم می‌تواند عدد بیرون از بازه تایپ کند. برای همین در JavaScript دوباره بازه را بررسی کرده‌ایم. اول هم خالی بودن کادر را بررسی کردیم، چون Number("") صفر می‌شود و بدون این شرط، کادر خالی «مردود» نشان داده می‌شد.



عملگر سه‌تایی (Ternary) — if کوتاه



// شرط ? مقدار اگر درست : مقدار اگر غلط
let age = 20;
let status = age >= 18 ? "بزرگسال" : "کودک";
console.log(status);                              // بزرگسال

let stock = 0;
console.log(stock > 0 ? "موجود" : "ناموجود");    // ناموجود










شکل کلی این عملگر سه بخش دارد: شرط، مقداری که اگر شرط درست باشد انتخاب می‌شود، و مقداری که اگر غلط باشد. نتیجه را می‌توان در متغیر ریخت یا مستقیم چاپ کرد. این عملگر دستور اضافه اجرا نمی‌کند؛ فقط یکی از دو مقدار را برمی‌گرداند. برای چند حالت یا چند خط دستور، if خواناتر می‌ماند.



عملگر سه‌تایی برای انتخاب بین دو مقدار عالی است. اگر چند حالت دارید یا باید چند دستور اجرا کنید، از همان if...else استفاده کنید تا کد خوانا بماند.



اشتباه رایج: گذاشتن سمی‌کالن بعد از پرانتز شرط: if (x > 5); { ... }. در این حالت بلوک همیشه اجرا می‌شود.



تمرین: (۱) برنامه‌ای بنویسید که ساعت فعلی را بگیرد (new Date().getHours() عددی بین ۰ تا ۲۳ برمی‌گرداند) و بر اساس آن پیام مناسب را داخل صفحه نشان دهد: ساعت ۵ تا ۱۱ «صبح بخیر»، ۱۲ تا ۱۴ «ظهر بخیر»، ۱۵ تا ۱۹ «عصر بخیر» و بقیه ساعت‌ها «شب بخیر». (۲) با یک input سال تولد شمسی کاربر را بگیرید، سن او را حساب کنید (سال جاری را ۱۴۰۵ در نظر بگیرید) و اگر ۱۸ سال یا بیشتر بود بنویسید «می‌توانید ثبت‌نام کنید»، وگرنه بنویسید چند سال دیگر باید صبر کند.
