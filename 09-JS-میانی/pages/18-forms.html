۱۸) فرم‌ها و اعتبارسنجی (Forms & Validation)

برای کار با فرم، رویداد submit را روی خود form می‌گیریم (نه روی دکمه). مرورگر به‌طور پیش‌فرض بعد از submit صفحه را دوباره بارگذاری می‌کند؛ با preventDefault() جلویش را می‌گیریم.

<form id="signup" novalidate>
  <label for="su-name">نام کامل:</label>
  <input type="text" id="su-name" name="fullname" autocomplete="name">

  <label for="su-email">ایمیل:</label>
  <input type="email" id="su-email" name="email" autocomplete="email" dir="ltr">

  <label for="su-pass">رمز عبور:</label>
  <input type="password" id="su-pass" name="password" autocomplete="new-password">

  <label for="su-pass2">تکرار رمز عبور:</label>
  <input type="password" id="su-pass2" name="password2" autocomplete="new-password">

  <label for="su-level">سطح:</label>
  <select id="su-level" name="level">
    <option value="">سطح خود را انتخاب کنید</option>
    <option value="basic">پایه</option>
    <option value="mid">میانی</option>
  </select>

  <label><input type="checkbox" name="rules"> قوانین را می‌پذیرم</label>
  <button type="submit">ثبت‌نام</button>
</form>
<ul id="errors" aria-live="polite"></ul>

این فرم فیلدهای نام، ایمیل، دو رمز، سطح و پذیرش قوانین را دارد. for هر برچسب با id فیلد یکی است تا کلیک روی نوشته هم کادر را فعال کند. name همان کلیدی است که بعداً از form.elements خوانده می‌شود. novalidate اعتبارسنجی آمادهٔ مرورگر را خاموش می‌کند تا قانون‌ها را خود برنامه با JavaScript بررسی کند. فهرست errors جای پیام‌هاست.

const form = document.getElementById("signup");
const errorList = document.getElementById("errors");

form.addEventListener("submit", (event) => {
  event.preventDefault();                  // صفحه دوباره بارگذاری نشود

  const name = form.elements.fullname.value.trim();
  const email = form.elements.email.value.trim();
  const password = form.elements.password.value;    // رمز را trim نمی‌کنیم
  const password2 = form.elements.password2.value;
  const level = form.elements.level.value;
  const rules = form.elements.rules.checked;

  const errors = [];
  if (name.length < 3) errors.push("نام باید حداقل ۳ حرف باشد.");
  if (!email.includes("@") || !email.includes(".")) errors.push("ایمیل معتبر نیست (مثال: ali@example.com).");
  if (password.length < 8) errors.push("رمز عبور باید حداقل ۸ کاراکتر باشد.");
  if (password !== password2) errors.push("رمز عبور و تکرار آن یکسان نیستند.");
  if (level === "") errors.push("سطح را انتخاب کنید.");
  if (!rules) errors.push("پذیرفتن قوانین الزامی است.");

  errorList.textContent = "";              // پیام‌های قبلی پاک شود
  if (errors.length > 0) {
    errors.forEach(message => {
      const li = document.createElement("li");
      li.className = "msg-error";
      li.textContent = message;
      errorList.append(li);
    });
  } else {
    const li = document.createElement("li");
    li.className = "msg-ok";
    li.textContent = `✔ ${name} عزیز، ثبت‌نام انجام شد.`;   // متن کاربر ← textContent
    errorList.append(li);
    form.reset();
  }
});

submit هم با دکمه و هم با Enter در کادر اجرا می‌شود. preventDefault جلوی بارگذاری دوبارهٔ صفحه را می‌گیرد تا پیام خطا پاک نشود. مقدار فیلدها از روی name خوانده می‌شود. نام و ایمیل فاصله‌های اضافه را از دست می‌دهند، ولی رمز بدون trim می‌ماند چون فاصله می‌تواند بخشی از رمز باشد. هر شرط نامعتبر یک جمله به آرایهٔ خطا اضافه می‌کند. اگر خطایی باشد، هر جمله یک li می‌شود؛ وگرنه پیام موفقیت با نام کاربر ساخته می‌شود و form.reset فیلدها را خالی می‌کند.







    form.elements.fullname یعنی فیلدی از فرم که name="fullname" دارد؛ form.elements همه فیلدهای فرم را بر اساس name در اختیار می‌گذارد (شکل کوتاه‌تر form.fullname هم در مرورگرها کار می‌کند، ولی اگر اسم فیلد با یک ویژگی خود فرم مثل action یکی باشد، مشکل‌ساز می‌شود). بررسی ایمیل را ساده گرفته‌ایم: فقط اینکه @ و نقطه داشته باشد.

    چرا پیام‌ها را با createElement ساختیم و نه innerHTML؟ پیام‌های خطا متن خودمان است و innerHTML هم کار می‌کرد، ولی عادت کردن به textContent باعث می‌شود وقتی متن کاربر را نشان می‌دهید (مثل پیام موفقیت که نام کاربر را دارد) دچار XSS نشوید.

    خواندن مقدار انواع فیلد

    فیلدخواندن مقدار
    text / email / password / textareainput.value
    numberNumber(input.value) یا input.valueAsNumber
    checkboxinput.checked (true/false)
    radioform.querySelector('input[name="gender"]:checked').value

    همه فیلدها با همnew FormData(form) و Object.fromEntries(...)

    اعتبارسنجی زنده هنگام تایپ

    const passInput = document.getElementById("pass-live");
const strength = document.getElementById("strength");

passInput.addEventListener("input", () => {
  const len = passInput.value.length;
  if (len === 0) {
    strength.textContent = "";
  } else if (len < 8) {
    strength.textContent = `ضعیف (${len} از حداقل ۸ کاراکتر)`;
    strength.className = "msg-error";
  } else {
    strength.textContent = "قابل قبول";
    strength.className = "msg-ok";
  }
});

    رویداد input با هر تغییر کادر اجرا می‌شود، نه فقط هنگام خروج از آن. طول صفر پیام را پاک می‌کند. کمتر از ۸ نویسه پیام خطا و کلاس قرمز می‌گیرد. از ۸ به بالا پیام قابل‌قبول و کلاس دیگر جایگزین کلاس قبلی می‌شود، چون className کل کلاس‌ها را عوض می‌کند نه اینکه یکی اضافه کند.

    رویداد input با هر تغییر کادر (تایپ، پاک کردن، چسباندن) اجرا می‌شود؛ ولی رویداد change برای کادر متنی فقط وقتی اجرا می‌شود که کاربر از کادر خارج شود.

    ویژگی‌های HTML مثل required، minlength و type="email" هم اعتبارسنجی آماده دارند. ما اینجا با novalidate آن‌ها را خاموش کردیم تا اعتبارسنجی را خودمان با JavaScript بنویسیم.

    اعتبارسنجی در مرورگر فقط برای راحتی کاربر است. سرور هم باید همیشه داده را دوباره بررسی کند.

    اشتباه رایج: گذاشتن رویداد click روی دکمه به جای submit روی فرم. با Enter زدن در کادر هم فرم ارسال می‌شود و رویداد click را از دست می‌دهید.

    اشتباه رایج: مقایسه مقدار کادر عددی بدون تبدیل: input.value > 5. مقدار value همیشه رشته است (حتی برای type="number") و "10" + 5 می‌شود "105"؛ اول با Number(...) تبدیل کنید.

    اشتباه رایج: فراموش کردن <label> و اکتفا به placeholder. placeholder با شروع تایپ ناپدید می‌شود و صفحه‌خوان‌ها همیشه آن را نمی‌خوانند؛ هر فیلد باید یک label با for برابر id خودش داشته باشد (کلیک روی label هم کادر را فعال می‌کند).

    تمرین: یک فرم «تماس با ما» با نام، ایمیل و پیام (textarea) و برای هر کدام یک label بسازید. هنگام submit: (الف) اگر نام خالی بود، (ب) ایمیل @ یا نقطه نداشت یا (ج) پیام کمتر از ۱۰ حرف بود، پیام خطای مناسب را زیر فرم نشان دهید. وگرنه اطلاعات را با Object.fromEntries(new FormData(form)) در Console چاپ کنید، پیام «پیام شما ارسال شد» را نشان دهید و فرم را خالی کنید.
