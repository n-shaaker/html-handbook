۱۴) عبارت‌های منظم (Regular Expressions)

عبارت منظم (RegExp) یک «الگو» برای جستجو در متن است؛ مثلاً «۱۱ رقم که با ۰۹ شروع شود». بین دو اسلش نوشته می‌شود: /الگو/پرچم. کاربرد اصلی‌اش در وب، اعتبارسنجی فرم (موبایل، ایمیل، کد ملی و ...) و جستجو و جایگزینی در متن است.

نمادمعنیمثال
\dیک رقم انگلیسی (0 تا 9). ارقام فارسی ۰ تا ۹ را شامل نمی‌شود!/\d\d/ دو رقم
\wحرف انگلیسی، رقم انگلیسی یا _ (حروف فارسی را شامل نمی‌شود)/\w+/
\sفاصله سفید: space، tab، خط جدید/\s+/
.هر کاراکتری به جز خط جدید/a.c/
[abc]یکی از a، b یا c/[aeiou]/
[^abc]هر چیزی به جز a، b و c/[^0-9]/ غیر رقم
[0-9]بازه/[a-z]/، /[۰-۹]/ رقم فارسی
+ / * / ?یک یا بیشتر / صفر یا بیشتر / صفر یا یک (اختیاری)/\d+/
{n} / {n,m} / {n,}دقیقاً n بار / n تا m بار / حداقل n بار/\d{4}/
^ / $شروع / پایان متن/^09\d{9}$/
a|ba یا b/jpg|png/
( )گروه‌بندی (و ذخیره بخش پیداشده)/(\d{4})-(\d{2})/
\.خود کاراکتر نقطه (کاراکترهای خاص با \ عادی می‌شوند)/\.ir$/
پرچممعنی
gهمه موارد (نه فقط اولی)
iبدون توجه به حروف بزرگ و کوچک
mچندخطی: ^ و $ شروع و پایان هر خط را می‌گیرند
uحالت یونیکد؛ برای کار درست با ایموجی و برخی کاراکترهای خاص

متدهای اصلی

const text = "قیمت‌ها: 120 و 4500 و 78 تومان";

console.log(/\d+/.test(text));            // true ← آیا عدد دارد؟
console.log(text.match(/\d+/));           // ["120"] ← اولین مورد (همراه با index و ...)
console.log(text.match(/\d+/g));          // ["120","4500","78"] ← همه موارد
console.log(text.replace(/\d+/g, "؟"));   // قیمت‌ها: ؟ و ؟ و ؟ تومان
console.log("JavaScript".search(/script/i));   // 4 ← جایگاه اولین مورد
console.log("a, b;c  d".split(/[,;\s]+/));     // ["a","b","c","d"]

test فقط می‌گوید الگو در متن هست یا نه. match بدون پرچم g اولین مورد و اطلاعات اضافه را برمی‌گرداند و با g همهٔ موردها را در یک آرایه می‌دهد. replace موردهای مطابق را با متن تازه عوض می‌کند. search شمارهٔ شروع اولین مورد است و split متن را از روی جداکننده‌هایی که الگو می‌گوید می‌بُرد.






مشکل ارقام فارسی



کاربر ایرانی ممکن است شماره موبایل را با صفحه‌کلید فارسی (۰۹۱۲...) یا عربی (٠٩١٢...) وارد کند. \d این ارقام را نمی‌شناسد؛ پس قبل از اعتبارسنجی، ارقام را به انگلیسی تبدیل می‌کنیم:



function toEnglishDigits(str) {
  return str
    .replace(/[۰-۹]/g, d => "۰۱۲۳۴۵۶۷۸۹".indexOf(d))    // ارقام فارسی
    .replace(/[٠-٩]/g, d => "٠١٢٣٤٥٦٧٨٩".indexOf(d));   // ارقام عربی
}

console.log(/^\d+$/.test("۰۹۱۲"));                 // false ← \d رقم فارسی را نمی‌شناسد
console.log(toEnglishDigits("۰۹۱۲۱۲۳۴۵۶۷"));       // 09121234567



کلاس \d در JavaScript رقم‌های فارسی و عربی را رقم حساب نمی‌کند. این تابع هر رقم فارسی را با جایگاهش در رشتهٔ «۰۱۲۳۴۵۶۷۸۹» به رقم انگلیسی متناظر تبدیل می‌کند و همین کار را برای رقم عربی تکرار می‌کند. بعد از این تبدیل می‌شود الگوهای عددی معمولی را به کار برد.










الگوهای کاربردی ایرانی



const mobile = /^09\d{9}$/;                       // 09 و ۹ رقم دیگر (جمعاً ۱۱ رقم)
const mobileIntl = /^(\+98|0)9\d{9}$/;            // با پیشوند 0 یا +98
const postalCode = /^\d{5}-?\d{5}$/;              // ۱۰ رقم؛ خط تیره وسط اختیاری
const email = /^[\w.-]+@[\w.-]+\.[a-z]{2,}$/i;    // نام @ دامنه . پسوند

console.log(mobile.test("09121234567"));              // true
console.log(mobile.test("0912123"));                  // false ← کوتاه است
console.log(mobileIntl.test("+989121234567"));        // true
console.log(postalCode.test("13185-34567"));          // true
console.log(email.test("sara@example.com"));          // true
console.log(email.test("ali.rezaei@mail.ut.ac.ir"));  // true ← زیردامنه هم مجاز است
console.log(email.test("sara@com"));                  // false ← پسوند دامنه ندارد



علامت ^ ابتدای متن و $ انتهای آن است، پس الگو باید کل رشته را پوشش دهد نه یک تکه را. \d{9} دقیقاً نه رقم است. الگوی بین‌المللی یکی از پیشوندهای +98 یا 0 را می‌پذیرد. کد پستی خط تیره را اختیاری کرده و ایمیل با پرچم i به بزرگی و کوچکی حروف حساس نیست. این الگوها نمونهٔ آموزشی‌اند و هر نشانی واقعی دنیا را پوشش نمی‌دهند.










در (\+98|0) پرانتز دو حالت پیشوند را گروه می‌کند و \+ یعنی خود کاراکتر +. الگوی ایمیل بالا برای فرم‌ها کافی است، ولی هیچ RegExp ساده‌ای همه ایمیل‌های معتبر را پوشش نمی‌دهد؛ تنها راه مطمئن، فرستادن ایمیل تأیید است. الگوی کد پستی هم فقط «قالب» (۱۰ رقم) را بررسی می‌کند، نه وجود واقعی آن را.



کد ملی: الگو کافی نیست!



کد ملی ایران ۱۰ رقم است، اما هر ۱۰ رقمی کد ملی معتبر نیست: رقم آخر یک «رقم کنترل» است که از روی ۹ رقم اول حساب می‌شود. پس RegExp فقط قالب را بررسی می‌کند و بقیه کار با کد معمولی انجام می‌شود:



function isValidNationalCode(input) {
  const code = toEnglishDigits(input.trim());
  if (!/^\d{10}$/.test(code)) return false;        // دقیقاً ۱۰ رقم
  if (code === code[0].repeat(10)) return false;   // ده رقم یکسان (مثل 1111111111) قابل قبول نیست

  const check = Number(code[9]);                   // رقم کنترل (آخرین رقم)
  let sum = 0;
  for (let i = 0; i < 9; i++) {
    sum += Number(code[i]) * (10 - i);             // رقم اول × 10، دوم × 9، ...، نهم × 2
  }
  const r = sum % 11;
  if (r < 2) {
    return check === r;
  } else {
    return check === 11 - r;
  }
}

console.log(isValidNationalCode("0012345679"));    // true
console.log(isValidNationalCode("0012345678"));    // false ← رقم کنترل اشتباه
console.log(isValidNationalCode("1111111111"));    // false
console.log(isValidNationalCode("۰۰۱۲۳۴۵۶۷۹"));    // true ← ارقام فارسی هم پذیرفته شد



اول رقم‌های فارسی به انگلیسی تبدیل می‌شوند و طول باید دقیقاً ده رقم باشد. ده رقم یکسان رد می‌شود. نه رقم اول هر کدام در وزن خودش ضرب و جمع می‌شوند. باقیماندهٔ تقسیم این جمع بر ۱۱ اگر کمتر از ۲ باشد باید با رقم آخر برابر باشد؛ در غیر این صورت رقم آخر باید برابر ۱۱ منهای همان باقیمانده باشد. اگر این برابری برقرار نباشد تابع false برمی‌گرداند، حتی وقتی طول کد درست است.










code[0].repeat(10) رقم اول را ده بار پشت هم می‌گذارد (مثلاً "1111111111")؛ اگر کد ملی با آن برابر بود، یعنی همه رقم‌ها یکسان‌اند.



مثال زنده: اعتبارسنجی هم‌زمان فرم



function validate(input, check, msgEl) {
  input.addEventListener("input", () => {
    const value = toEnglishDigits(input.value.trim());
    if (value === "") {
      msgEl.textContent = "";
      input.removeAttribute("aria-invalid");
      return;
    }
    const ok = check(value);
    msgEl.textContent = ok ? "✔ معتبر" : "✖ نامعتبر";
    msgEl.className = ok ? "msg-ok" : "msg-error";
    input.setAttribute("aria-invalid", !ok);       // برای صفحه‌خوان‌ها
  });
}

validate(mobileInput, v => mobile.test(v), mobileMsg);
validate(emailInput, v => email.test(v), emailMsg);
validate(nationalInput, isValidNationalCode, nationalMsg);
validate(postalInput, v => postalCode.test(v), postalMsg);



این تابع شنوندهٔ input را روی یک کادر نصب می‌کند و بررسی را به تابع check می‌سپارد. کادر خالی پیام را پاک می‌کند تا قبل از تایپ خطا نشان داده نشود. نتیجهٔ بررسی، متن و کلاس پیام را عوض می‌کند و aria-invalid را برای ابزارهای کمکی تنظیم می‌کند. هر کادر می‌تواند تابع بررسی خودش را داشته باشد.




 

 

 

 




ارقام فارسی را هم امتحان کنید (مثلاً ۰۹۱۲۱۲۳۴۵۶۷)؛ چون تابع validate اول toEnglishDigits را صدا می‌زند، پذیرفته می‌شوند. یادتان باشد اعتبارسنجی سمت مرورگر فقط برای راحتی کاربر است؛ سرور هم باید داده را دوباره بررسی کند.



برای تست و یادگیری RegExp از سایت regex101.com استفاده کنید (گزینه Flavor را روی ECMAScript/JavaScript بگذارید)؛ هر بخش الگو را توضیح می‌دهد.



اشتباه رایج: فراموش کردن ^ و $ در اعتبارسنجی. بدون آن‌ها /09\d{9}/ متن abc09121234567xyz را هم قبول می‌کند.



اشتباه رایج: استفاده از پرچم g همراه با test روی یک الگوی ذخیره‌شده در متغیر. الگوی g «جای آخرین جستجو» (lastIndex) را به خاطر می‌سپارد؛ برای همین با const re = /\d/g; دو بار پشت سر هم re.test("a1") اول true و بعد false می‌دهد! برای اعتبارسنجی پرچم g لازم نیست.



اشتباه رایج: فراموش کردن \ قبل از نقطه: /\.ir$/ یعنی «با .ir تمام شود»، اما /.ir$/ هر کاراکتری قبل از ir را قبول می‌کند (مثلاً "amir").



تمرین:

۱) تابع isStrongPassword(pass) بنویسید که بررسی کند رمز عبور حداقل ۸ کاراکتر، حداقل یک رقم و حداقل یک حرف انگلیسی دارد (راهنما: سه تست جدا با /.{8,}/، /\d/ و /[a-z]/i). یک input رمز بسازید و قدرت رمز را هم‌زمان نشان دهید.

۲) الگویی بنویسید که فقط نام فارسی (حروف فارسی، فاصله و نیم‌فاصله) را بپذیرد و آن را با «علی‌رضا محمدی» (معتبر) و «Ali» (نامعتبر) امتحان کنید (راهنما: /^[آ-ی\s\u200c]+$/؛ \u200c همان نیم‌فاصله است).

۳) با replace و یک الگوی g، همه فاصله‌های تکراری یک متن را به یک فاصله تبدیل کنید.
