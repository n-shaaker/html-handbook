۲۰) پروژه نهایی: فروشگاه کوچک

این پروژه مباحث هر سه دوره را کنار هم می‌گذارد: کلاس، آرایه و متدهایش، Promise و async/await (بارگذاری شبیه‌سازی‌شده محصولات)، Event Delegation، localStorage (سبد خرید ماندگار) و ... . اول نسخه زنده را امتحان کنید؛ سبد خرید با بارگذاری دوباره صفحه پاک نمی‌شود.

🛒 سبد خرید

    ۱) مدل داده با کلاس

    class Product {
  constructor(id, name, price, category) {
    this.id = id;
    this.name = name;
    this.price = price;
    this.category = category;
  }

  priceText() {
    return this.price.toLocaleString("fa-IR") + " تومان";
  }
}

    کلاس محصول دادهٔ ثابت هر کالا را نگه می‌دارد: شناسه، نام، قیمت و دسته. متد priceText قیمت را با رقم فارسی و واحد پول به متن نمایش تبدیل می‌کند تا همین شکل در فهرست و سبد تکرار نشود. ساختن عنصر صفحه داخل این کلاس نیست.

    ۲) سبد خرید با آرایه و localStorage

    // سبد خرید: آرایه‌ای از { id, qty } که در localStorage ذخیره می‌شود
class Cart {
  constructor(storageKey) {
    this.key = storageKey;
    this.items = this.load();
  }

  load() {
    try {
      const text = localStorage.getItem(this.key);
      return text ? JSON.parse(text) : [];
    } catch (err) {
      return [];        // داده خراب بود ← سبد خالی، بدون از کار افتادن برنامه
    }
  }

  save() {
    localStorage.setItem(this.key, JSON.stringify(this.items));
  }

  add(id) {
    const item = this.items.find(item => item.id === id);
    if (item) {
      item.qty++;
    } else {
      this.items.push({ id: id, qty: 1 });
    }
    this.save();
  }

  remove(id) {
    const item = this.items.find(item => item.id === id);
    if (!item) return;
    item.qty--;
    if (item.qty === 0) {
      this.items = this.items.filter(item => item.id !== id);
    }
    this.save();
  }

  clear() {
    this.items = [];
    this.save();
  }
}

    سبد فقط شناسه و تعداد را نگه می‌دارد، نه کل اطلاعات محصول. load متن ذخیره‌شده را می‌خواند و اگر خالی یا خراب باشد آرایهٔ خالی برمی‌گرداند تا برنامه شروع شود. هر تغییری که تعداد را عوض کند باید دوباره در localStorage ذخیره شود تا با تازه‌کردن صفحه از بین نرود.

    سبد فقط id و تعداد هر کالا را نگه می‌دارد، نه خود محصول را؛ اسم و قیمت را موقع نمایش از آرایه products پیدا می‌کنیم. بعد از هر تغییر، save() کل آرایه را با JSON.stringify در localStorage می‌نویسد و موقع ساختن سبد، load() آن را برمی‌گرداند.

    ۳) بارگذاری ناهمگام محصولات

    function wait(ms) {
  return new Promise(resolve => setTimeout(resolve, ms));
}

async function fetchProducts() {
  await wait(800);   // شبیه‌سازی تأخیر شبکه؛ در پروژه واقعی: await fetch(url)
  return PRODUCTS_DATA.map(p => new Product(p.id, p.name, p.price, p.category));
}

    wait یک Promise است که بعد از مدت مشخص resolve می‌شود و اینجا جای تأخیر شبکه را گرفته است. fetchProducts بعد از آن انتظار، هر شیء خام را با new Product به نمونهٔ کلاس تبدیل می‌کند. در برنامهٔ واقعی، به‌جای wait از fetch و response.json استفاده می‌کنید.

    ۴) نمایش، جستجو، مرتب‌سازی و Event Delegation

    function getVisibleProducts() {
  const q = searchInput.value.trim();
  const result = products.filter(p => p.name.includes(q));

  if (sortSelect.value === "price-asc") result.sort((a, b) => a.price - b.price);
  if (sortSelect.value === "price-desc") result.sort((a, b) => b.price - a.price);
  if (sortSelect.value === "name") result.sort((a, b) => a.name.localeCompare(b.name, "fa"));

  return result;
}

function renderProducts() {
  productsBox.innerHTML = "";
  const visible = getVisibleProducts();
  if (visible.length === 0) {
    productsBox.textContent = "محصولی پیدا نشد.";
    return;
  }

  visible.forEach(p => {
    const card = document.createElement("div");
    card.className = "js-card";

    const title = document.createElement("h4");
    title.textContent = p.name;

    const info = document.createElement("p");
    info.textContent = p.category + " — " + p.priceText();

    const btn = document.createElement("button");
    btn.className = "btn";
    btn.textContent = "افزودن به سبد";
    btn.dataset.add = p.id;          // در HTML: data-add="1"

    card.append(title, info, btn);
    productsBox.append(card);
  });
}

function renderCart() {
  cartList.innerHTML = "";
  let total = 0;

  cart.items.forEach(item => {
    const p = products.find(p => p.id === item.id);
    if (!p) return;                  // محصولی که دیگر وجود ندارد

    const sum = p.price * item.qty;
    total += sum;

    const li = document.createElement("li");
    const text = document.createElement("span");
    text.textContent = `${p.name} × ${item.qty} = ${sum.toLocaleString("fa-IR")}`;

    const btn = document.createElement("button");
    btn.className = "btn danger";
    btn.textContent = "−";
    btn.dataset.remove = p.id;
    btn.setAttribute("aria-label", "کم کردن یکی از " + p.name);

    li.append(text, btn);
    cartList.append(li);
  });

  if (cartList.children.length === 0) {
    cartList.innerHTML = "<li>سبد خالی است</li>";
  }
  totalText.textContent = `جمع کل: ${total.toLocaleString("fa-IR")} تومان`;
}

// یک listener برای همه دکمه‌های «افزودن»
productsBox.addEventListener("click", e => {
  const id = e.target.dataset.add;
  if (!id) return;                   // روی دکمه «افزودن» کلیک نشده
  cart.add(Number(id));
  renderCart();
});

// یک listener برای همه دکمه‌های «−»
cartList.addEventListener("click", e => {
  const id = e.target.dataset.remove;
  if (!id) return;
  cart.remove(Number(id));
  renderCart();
});

searchInput.addEventListener("input", renderProducts);
sortSelect.addEventListener("change", renderProducts);
clearBtn.addEventListener("click", () => {
  cart.clear();
  renderCart();
});

    getVisibleProducts نام را با متن جست‌وجو فیلتر می‌کند و در صورت نیاز همان نتیجه را با قیمت یا نام مرتب می‌کند. renderProducts فهرست دیده‌شده را به عنصرهای صفحه تبدیل می‌کند. رویدادهای جست‌وجو و مرتب‌سازی دادهٔ اصلی products را خراب نمی‌کنند؛ هر بار از همان آرایه نمای تازه می‌سازند.

    نام و قیمت محصولات با textContent در صفحه گذاشته می‌شود، نه با innerHTML. این‌طوری اگر روزی داده از کاربر یا یک API ناشناس بیاید و داخلش کد HTML باشد، اجرا نمی‌شود (جلوی حمله XSS گرفته می‌شود).

    ۵) شروع برنامه

    const cart = new Cart("shop-cart");
let products = [];

async function init() {
  statusEl.textContent = "در حال بارگذاری محصولات...";
  try {
    products = await fetchProducts();
    statusEl.textContent = `${products.length} محصول بارگذاری شد.`;
    renderProducts();
    renderCart();
  } catch (err) {
    statusEl.textContent = "خطا در بارگذاری: " + err.message;
  }
}

init();

    init اول پیام بارگذاری را نشان می‌دهد، بعد محصولات را await می‌کند و هم فهرست کالا و هم سبد را رسم می‌کند. اگر دریافت داده خطا بدهد، catch پیام را در همان عنصر وضعیت می‌نویسد و صفحه سفید نمی‌ماند. صدا زدن init در پایان فایل، شروع واقعی برنامه است.

    تمرین‌های تکمیلی

    1. فیلتر بر اساس دسته‌بندی با یک select اضافه کنید (فهرست دسته‌ها را با [...new Set(products.map(p => p.category))] بسازید و گزینه «همه» هم داشته باشد).
    2. جستجو را با debounce (درس ۷) انجام دهید تا با هر حرف، فهرست دوباره ساخته نشود.
    3. به جای داده ثابت، محصولات را با تابع fetchJSON (درس ۱۱) از https://dummyjson.com/products?limit=10 بگیرید. دقت کنید پاسخ این API شیئی است که آرایه محصولات در ویژگی products آن قرار دارد و نام محصول در title است؛ پس باید داده را به شکل { id, name, price, category } تبدیل کنید. پیام بارگذاری و خطای فارسی را هم فراموش نکنید.
    4. کد تخفیف اضافه کنید: اگر کاربر OFF20 وارد کرد (بدون توجه به حروف بزرگ و کوچک)، ۲۰٪ از جمع کم شود و جمع قبل و بعد از تخفیف هر دو نمایش داده شوند.
    5. کد را به ماژول‌ها تقسیم کنید: product.js، cart.js، ui.js و app.js (و صفحه را با Live Server اجرا کنید).
    6. یک «حالت شب» با ذخیره انتخاب کاربر در localStorage اضافه کنید (درس ۱۳).
    7. دکمه «+» کنار هر کالای سبد بگذارید و همه دکمه‌ها را همچنان با همان یک listener روی cartList مدیریت کنید.

    قدم بعدی: حالا که JavaScript را تا سطح پیشرفته یاد گرفته‌اید، آماده یادگیری یک فریم‌ورک مثل React یا Vue هستید. مفاهیمی مثل state، render، کلاس، ماژول و async در همه آن‌ها تکرار می‌شوند.
