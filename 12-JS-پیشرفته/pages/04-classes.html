۴) کلاس‌ها (Classes)

کلاس یک قالب (نقشه) برای ساختن اشیای مشابه است. از روی یک کلاس می‌توان هر تعداد شیء (نمونه / instance) ساخت.

class Car {
  constructor(brand, year) {      // موقع ساختن شیء اجرا می‌شود
    this.brand = brand;
    this.year = year;
  }

  age() {                         // متد
    return new Date().getFullYear() - this.year;
  }

  info() {
    return `${this.brand} مدل ${this.year} (${this.age()} ساله)`;
  }
}

const car1 = new Car("پژو", 2015);
const car2 = new Car("تویوتا", 2020);

console.log(car1.info());            // پژو مدل 2015 (... ساله)
console.log(car2.info());            // تویوتا مدل 2020 (... ساله)
console.log(car1 instanceof Car);    // true

constructor هنگام new اجرا می‌شود و ویژگی‌های همان نمونه را روی this می‌گذارد. age و info متد هستند و برای هر نمونه جدا کپی نمی‌شوند؛ از روی شیء ساخته‌شده صدا زده می‌شوند. instanceof بررسی می‌کند شیء با آن کلاس ساخته شده است یا نه.






سن خودرو از روی سال جاری سیستم حساب می‌شود؛ پس عددی که شما می‌بینید ممکن است با هم‌کلاسی‌تان در سال دیگر فرق کند. متدهایی مثل age و info یک بار روی Car.prototype ساخته می‌شوند و همه نمونه‌ها از همان یک نسخه استفاده می‌کنند.

getter و setter



getter (با get) مقداری را «محاسبه» می‌کند ولی مثل یک ویژگی معمولی و بدون پرانتز خوانده می‌شود. setter (با set) موقع مقداردهی با = اجرا می‌شود و جای خوبی برای تبدیل یا اعتبارسنجی مقدار است.



class Temperature {
  constructor(celsius) {
    this.celsius = celsius;
  }

  get fahrenheit() {                 // مثل ویژگی خوانده می‌شود، نه با ()
    return this.celsius * 1.8 + 32;
  }

  set fahrenheit(f) {
    this.celsius = (f - 32) / 1.8;
  }
}

const t = new Temperature(25);
console.log(t.fahrenheit);    // 77   ← getter اجرا شد
t.fahrenheit = 212;           // setter اجرا شد
console.log(t.celsius);       // 100



getter مثل ویژگی خوانده می‌شود و پرانتز نمی‌خواهد، ولی پشت آن یک تابع است که هر بار مقدار را حساب می‌کند. setter وقتی به همان نام مقدار می‌دهید اجرا می‌شود و اینجا فارنهایت را به سلسیوس برمی‌گرداند و در celsius ذخیره می‌کند. دادهٔ اصلی همچنان سلسیوس است.










اشتباه رایج: هم‌نام بودن setter و ویژگی‌ای که داخلش مقدار می‌دهید: set name(v) { this.name = v; }. این خط دوباره همان setter را صدا می‌زند و خطای Maximum call stack size exceeded می‌گیرید. مقدار واقعی را در ویژگی دیگری (مثلاً this._name) یا در یک فیلد خصوصی #name (درس بعد) نگه دارید.



مثال کاربردی: کلاس سبد خرید



class ShoppingCart {
  constructor() {
    this.items = [];
  }

  add(name, price, qty = 1) {
    const found = this.items.find(item => item.name === name);
    if (found) found.qty += qty;            // کالای تکراری: فقط تعداد زیاد شود
    else this.items.push({ name, price, qty });
    return this;                            // برگرداندن this ← امکان زنجیره‌ای صدا زدن
  }

  remove(name) {
    this.items = this.items.filter(item => item.name !== name);
    return this;
  }

  get count() {
    return this.items.reduce((sum, item) => sum + item.qty, 0);
  }

  get total() {
    return this.items.reduce((sum, item) => sum + item.price * item.qty, 0);
  }
}

const cart = new ShoppingCart();
cart.add("کتاب", 250000).add("خودکار", 30000, 3).add("کتاب", 250000);
console.log(cart.count);                 // 5
console.log(cart.total);                 // 590000
cart.remove("خودکار");
console.log(cart.count, cart.total);     // 2 500000



add اگر کالا از قبل باشد فقط تعداد را زیاد می‌کند و وگرنه شیء تازه‌ای به آرایه اضافه می‌کند. برگرداندن this اجازه می‌دهد چند add را پشت سر هم بنویسید. count و total از روی آرایهٔ فعلی حساب می‌شوند، نه اینکه عدد جدایی باشند که ممکن است از قلم بیفتد.










مثال زنده: ساختن اشیا از روی کلاس



class Product {
  constructor(name, price) {
    this.name = name;
    this.price = price;
  }

  priceWithTax() {
    return Math.round(this.price * 1.1);
  }

  toCard() {
    const card = document.createElement("div");
    card.className = "js-card";
    const title = document.createElement("h4");
    title.textContent = this.name;
    const text = document.createElement("p");
    text.textContent = `قیمت با مالیات: ${this.priceWithTax().toLocaleString("fa-IR")} تومان`;
    card.append(title, text);
    return card;
  }
}

addBtn.addEventListener("click", () => {
  const price = Number(priceInput.value);
  if (priceInput.value === "" || !Number.isFinite(price) || price <= 0) {
    errorText.textContent = "قیمت باید یک عدد بزرگ‌تر از صفر باشد.";
    return;
  }
  errorText.textContent = "";
  const p = new Product(nameInput.value.trim() || "محصول", price);
  cards.append(p.toCard());
});



priceWithTax ده درصد را حساب می‌کند و با Math.round به تومان صحیح گرد می‌کند. toCard یک عنصر کارت می‌سازد، نام و قیمت با مالیات را داخلش می‌نویسد و خود عنصر را برمی‌گرداند. با کلیک، اگر کادر قیمت خالی، غیرعددی یا صفر و کمتر باشد پیام خطا می‌ماند. در غیر این صورت یک Product ساخته می‌شود و کارت برگشتی به صفحه اضافه می‌شود. اگر نام خالی باشد، متن «محصول» به‌جای نام به کار می‌رود.















در این مثال مالیات بر ارزش افزوده ۱۰٪ فرض شده است (price * 1.1). برای نمایش متن کاربر (نام محصول) از textContent استفاده شده تا اگر کاربر کد HTML وارد کرد، اجرا نشود.



کلاس در واقع شکل مرتب‌تر همان «تابع سازنده» و «prototype» است که در کدهای قدیمی‌تر می‌بینید.



اشتباه رایج: صدا زدن کلاس بدون new: Car("پژو", 2015) خطای Class constructor cannot be invoked without 'new' می‌دهد.



تمرین: کلاس Student با ویژگی‌های name و scores (آرایه نمره‌ها از ۰ تا ۲۰) بسازید. متد average() (میانگین نمره‌ها؛ برای آرایه خالی ۰) و getter به اسم status اضافه کنید که اگر میانگین ۱۰ یا بیشتر بود «قبول» و وگرنه «مردود» برگرداند. سپس متد addScore(n) بنویسید که نمره خارج از بازه ۰ تا ۲۰ را نپذیرد.
